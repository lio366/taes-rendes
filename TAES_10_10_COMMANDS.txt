# 3. Copy the files from this package into TAES.

# 6. Local start

# 7. Health
curl http://localhost:10000/health

curl -X POST http://localhost:10000/sandbox/run   -H "Content-Type: application/json"   -d '{"action":"sandbox_test"}'

# 9. Test agent
curl -X POST http://localhost:10000/agent/run   -H "Content-Type: application/json"   -d '{"task":"Preparar un plan para investigar una API publica"}'

# 10. Test capability registration
curl -X POST http://localhost:10000/capabilities/discover   -H "Content-Type: application/json"   -d '{"name":"example_api","kind":"api","endpoint":"https://example.com","license":"mit","cost":"free"}'

# 11. Test controlled HTTP
curl -X POST http://localhost:10000/tools/http   -H "Content-Type: application/json"   -d '{"method":"GET","url":"https://example.com"}'

# 12. Inspect jobs
curl http://localhost:10000/jobs

# 13. Inspect audit
curl http://localhost:10000/audit

# 14. Optional local full infrastructure
docker compose up --build

# 15. Verify
curl http://localhost:10000/health

git add .
